cmake_minimum_required(VERSION 3.22)
project(Fjord VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "")
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "")
endif()

# Plugin formats per platform: VST3 + Standalone everywhere, AU on macOS (Logic, GarageBand),
# LV2 on Linux (Ardour, Carla, Qtractor ...).
set(FJORD_FORMATS VST3 Standalone)
if(APPLE)
    list(APPEND FJORD_FORMATS AU)
elseif(UNIX)
    list(APPEND FJORD_FORMATS LV2)
endif()

# JUCE 8 is not part of this repository. A local checkout in external/JUCE (or -DFJORD_JUCE_DIR=<path>) is used
# if present; otherwise the pinned release is downloaded when CMake configures.
set(FJORD_JUCE_TAG "8.0.4")
set(FJORD_JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/JUCE" CACHE PATH "Local JUCE checkout (optional)")
if(EXISTS "${FJORD_JUCE_DIR}/CMakeLists.txt")
    add_subdirectory(${FJORD_JUCE_DIR} JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
                              GIT_TAG ${FJORD_JUCE_TAG} GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()

# Git commit shown in the About panel (empty outside a git checkout, e.g. in a source archive)
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
                OUTPUT_VARIABLE FJORD_GIT_HASH OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)

# Copy the built plugins to the user's plugin folders (turn off in CI: -DFJORD_COPY_PLUGIN=OFF)
option(FJORD_COPY_PLUGIN "Install the plugins to the user plugin folders after building" ON)

juce_add_plugin(Fjord
    COMPANY_NAME "Fjord"
    BUNDLE_ID "com.fjord.synth"
    PLUGIN_MANUFACTURER_CODE Fjrd
    PLUGIN_CODE Fjrd
    FORMATS ${FJORD_FORMATS}
    LV2URI "https://github.com/fjord-sounds/fjord"
    PRODUCT_NAME "Fjord"
    ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/resources/icon.png"
    VERSION ${PROJECT_VERSION}
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    AU_MAIN_TYPE kAudioUnitType_MusicDevice
    VST3_CATEGORIES Instrument Synth
    COPY_PLUGIN_AFTER_BUILD ${FJORD_COPY_PLUGIN})

file(GLOB_RECURSE FJORD_SOURCES CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/*.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/Source/*.h)
target_sources(Fjord PRIVATE ${FJORD_SOURCES})
target_include_directories(Fjord PRIVATE Source)

# Factory presets (generated by tools/preset-gen/make_presets.py) are embedded in the plugin.
file(GLOB_RECURSE FJORD_PRESETS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/presets/Factory/*.fjord
                                                  ${CMAKE_CURRENT_SOURCE_DIR}/presets/FxFactory/*.fjordfx
                                                  ${CMAKE_CURRENT_SOURCE_DIR}/presets/FxModules/*.fjordmod)
# Font: Manrope (SIL Open Font License, see resources/fonts/OFL.txt)
file(GLOB FJORD_FONTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/resources/fonts/*.ttf)
juce_add_binary_data(FjordBinaryData SOURCES ${FJORD_PRESETS} ${FJORD_FONTS})
set_target_properties(FjordBinaryData PROPERTIES POSITION_INDEPENDENT_CODE TRUE)

target_compile_definitions(Fjord PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0)
if(FJORD_GIT_HASH)
    target_compile_definitions(Fjord PUBLIC FJORD_GIT_HASH="${FJORD_GIT_HASH}")
endif()

target_link_libraries(Fjord
    PRIVATE FjordBinaryData juce::juce_audio_utils juce::juce_dsp
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)

# Offline render test (no DAW): cmake --build build --target FjordRenderTest
juce_add_console_app(FjordRenderTest PRODUCT_NAME "FjordRenderTest")
target_sources(FjordRenderTest PRIVATE tests/RenderTest.cpp ${FJORD_SOURCES})
target_include_directories(FjordRenderTest PRIVATE Source)
target_compile_definitions(FjordRenderTest PRIVATE
    JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JucePlugin_Name="Fjord" JucePlugin_VersionString="${PROJECT_VERSION}" JucePlugin_Build_Standalone=0)
target_link_libraries(FjordRenderTest PRIVATE
    FjordBinaryData juce::juce_audio_utils juce::juce_dsp juce::juce_audio_formats juce::juce_recommended_config_flags)
if(APPLE)
    set_target_properties(FjordRenderTest PROPERTIES OSX_ARCHITECTURES arm64)
endif()
